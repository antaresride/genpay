xcode-select --install || true

# Create working directory
mkdir -p ~/llvm-build && cd ~/llvm-build

# Clone LLVM 22.1.8 exact release tag
git clone --depth 1 --branch llvmorg-22.1.8 https://github.com/llvm/llvm-project.git

cd llvm-project

# Create build directory
mkdir -p build && cd build

# Configure with CMake
cmake -G Ninja \
  -S ../llvm \
  -B . \
  -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_INSTALL_PREFIX=$HOME/llvm-22 \
  -DCMAKE_OSX_ARCHITECTURES=arm64 \
  -DCMAKE_C_COMPILER=$(which clang) \
  -DCMAKE_CXX_COMPILER=$(which clang++) \
  -DDEFAULT_SYSROOT="$(xcrun --show-sdk-path)" \
  -DLLVM_DEFAULT_TARGET_TRIPLE=arm64-apple-darwin \
  -DLLVM_TARGETS_TO_BUILD="AArch64;X86;ARM;WebAssembly" \
  -DLLVM_ENABLE_PROJECTS="clang;lld" \
  -DLLVM_ENABLE_ASSERTIONS=ON \
  -DLLVM_ENABLE_RTTI=ON \
  -DLLVM_ENABLE_EH=ON \
  -DLLVM_BUILD_LLVM_DYLIB=ON \
  -DLLVM_LINK_LLVM_DYLIB=ON \
  -DLLVM_INSTALL_UTILS=ON \
  -DLLVM_ENABLE_ZLIB=OFF \
  -DLLVM_ENABLE_LIBXML2=OFF \
  -DLLVM_ENABLE_TERMINFO=OFF

# Build & Install
ninja
ninja install

# Append LLVM 22 configuration to ~/.zshrc
cat << 'EOF' >> ~/.zshrc

# LLVM 22 for Inkwell / llvm-sys 221
export LLVM_SYS_221_PREFIX=$HOME/llvm-22
export PATH=$HOME/llvm-22/bin:$PATH
export LIBRARY_PATH=$HOME/llvm-22/lib:$LIBRARY_PATH
EOF

# Apply updates to current session
source ~/.zshrc

# Verify binaries and version output
llvm-config --version    # Should display 22.1.0

# Ensure binary target path resolves correctly
which llvm-config        # Should point to $HOME/llvm-22/bin/llvm-config

# Verify RTTI remains enabled (Required by Inkwell)
llvm-config --has-rtti   # Should return "YES"

# Verify link targets
llvm-config --libs
llvm-config --ldflags
